232: function compareSkillKeySet(a: Record<string, any>, b: Record<string, any>): boolean {
233:   if (!a || !b) return false;

235:   const bKeys = Object.keys(b).sort();
236:   if (aKeys.length !== bKeys.length) return false;
237:   return aKeys.every((key, index) => key === bKeys[index]);
